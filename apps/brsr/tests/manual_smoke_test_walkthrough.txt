## STEP 2 — Paste into the shell: imports + look up existing data

## Everything from here on is Python, pasted into the `>>>` prompt.

from datetime import date
from django.contrib.contenttypes.models import ContentType
from django.contrib.auth import get_user_model
from apps.brsr.models import AssignmentSchedule, BRSRSection, Assignment
from apps.organizations.models import Plant, ApprovalConfigurationTemplate

User = get_user_model()

admin = User.objects.filter(is_superuser=True).first()
plant = Plant.objects.first()
section = BRSRSection.objects.first()
template = ApprovalConfigurationTemplate.objects.filter(framework="BRSR", is_active=True).first()

print("admin:", admin)
print("plant:", plant)
print("section:", section)
print("template:", template)
# If ANY of these print None, stop here — that's your actual blocker
# (e.g. no active BRSR ApprovalConfigurationTemplate exists yet for any
# company). Create the missing data via the admin site or another shell
# command before continuing.


## ============================================================
## STEP 3 — Pick a valid assignee (must satisfy the SAME eligibility
## rule as manual assignment: role matches the workflow's first stage's
## role, and the user is assigned to `plant`)
## ============================================================

first_stage = template.first_stage
print("first_stage:", first_stage, "| required role:", first_stage.role if first_stage else None)

# List candidates that already qualify:
eligible = User.objects.filter(
    is_active=True,
    assigned_plants=plant,
    role=first_stage.role,
) if first_stage and first_stage.role_id else User.objects.none()

print("eligible assignees:", list(eligible))

# If `eligible` is empty, either:
#   (a) pick any real user and wire them up manually, e.g.:
assignee = User.objects.exclude(pk=admin.pk).first()
assignee.role = first_stage.role
assignee.save(update_fields=["role"])
assignee.assigned_plants.add(plant)
#   (b) or just use one of the `eligible` users directly:
# assignee = eligible.first()

print("using assignee:", assignee)


## ============================================================
## STEP 4 — Create the test schedule (annual = "due" every day
## until one assignment exists for the FY, easiest for on-demand testing)
## ============================================================

schedule = AssignmentSchedule.objects.create(
    name="Manual smoke test",
    plant=plant,
    section=section,
    financial_year="2026-2027",
    workflow_template=template,
    frequency="annually",
    priority="medium",
    created_by=admin,
    assignee_content_type=ContentType.objects.get_for_model(User),
    assignee_object_id=assignee.pk,
)
schedule.questions.set(section.questions.filter(is_active=True)[:2])
print("schedule created:", schedule.schedule_id, "| questions:", schedule.questions.count())

if schedule.questions.count() == 0:
    print("WARNING: no active questions on this section — pick a different section "
          "or check BRSRQuestion.is_active for this section.")


## ============================================================
## STEP 5 — Confirm nothing has been generated yet
## ============================================================

print("assignments before generation:", Assignment.objects.filter(schedule=schedule).count())
# expect: 0


## ============================================================
## STEP 6 — Run generation (this is the exact function Celery Beat calls)
## ============================================================

from apps.brsr.services import run_daily_schedule_generation

created = run_daily_schedule_generation()
print("created:", created)
# expect: a non-empty list, e.g. [<Assignment: TST-2026-0007 -> assignee>]
#
# If this raises an exception instead, READ THE ERROR — likely causes:
#   - "No eligible assignee matches the first stage..." -> Step 3's role/
#     plant wiring didn't stick, or first_stage.role doesn't match.
#   - Any NotNullViolation / IntegrityError -> some required field on
#     Assignment/QuestionResponse isn't being filled; paste the traceback.


## ============================================================
## STEP 7 — Inspect what got created
## ============================================================

a = Assignment.objects.filter(schedule=schedule).first()
print("assignment_id:", a.assignment_id)
print("period_code:", a.period_code)
print("period_label:", a.period_label)
print("financial_year:", a.financial_year)
print("workflow_task:", a.workflow_task)          # should NOT be None
print("workflow_stage:", a.workflow_stage_label if a.workflow_task else None)
print("question_count:", a.questions.count())
print("responses:", list(a.responses.values_list("question__question_id", "status")))


## ============================================================
## STEP 8 — Confirm duplicate prevention (run generation again)
## ============================================================

created_again = run_daily_schedule_generation()
print("created_again:", created_again)
# expect: []

print("assignments after second run:", Assignment.objects.filter(schedule=schedule).count())
# expect: still 1

## ============================================================

exit()
